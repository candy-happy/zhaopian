function showToast(msg, type = 'info') {
  const c = document.getElementById('toast-container');
  const t = document.createElement('div');
  t.className = `toast ${type}`;
  t.textContent = msg;
  c.appendChild(t);
  setTimeout(() => { t.style.opacity = '0'; t.style.transform = 'translateX(40px)'; t.style.transition = 'all 0.3s'; }, 2500);
  setTimeout(() => t.remove(), 2800);
}

function showProcessing(text = '处理中...') {
  document.getElementById('processing-text').textContent = text;
  document.getElementById('processing').classList.add('show');
}
function hideProcessing() {
  document.getElementById('processing').classList.remove('show');
}

function downloadBlob(blob, filename) {
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = filename;
  a.click();
  URL.revokeObjectURL(a.href);
}

function loadImage(file) {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => resolve(img);
    img.onerror = reject;
    img.src = URL.createObjectURL(file);
  });
}

function formatSize(bytes) {
  if (bytes < 1024) return bytes + ' B';
  if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB';
  return (bytes / (1024 * 1024)).toFixed(1) + ' MB';
}

function extForMime(mime) {
  const map = { 'image/png': '.png', 'image/jpeg': '.jpg', 'image/webp': '.webp', 'image/bmp': '.bmp' };
  return map[mime] || '.png';
}

function initUploadZone(zoneEl, inputEl, onFiles) {
  zoneEl.addEventListener('click', () => inputEl.click());
  zoneEl.addEventListener('dragover', e => { e.preventDefault(); zoneEl.classList.add('dragover'); });
  zoneEl.addEventListener('dragleave', () => zoneEl.classList.remove('dragover'));
  zoneEl.addEventListener('drop', e => {
    e.preventDefault(); zoneEl.classList.remove('dragover');
    if (e.dataTransfer.files.length) onFiles(e.dataTransfer.files);
  });
  inputEl.addEventListener('change', () => {
    if (inputEl.files.length) onFiles(inputEl.files);
    inputEl.value = '';
  });
}

let cropImg = null;
let cropRatio = 0;

const cropCanvas = document.getElementById('crop-canvas');
const cropCtx = cropCanvas.getContext('2d');
const cropContainer = document.querySelector('.crop-preview-wrap');

// 输入框元素
const inputX = document.getElementById('crop-x');
const inputY = document.getElementById('crop-y');

const inputH = document.getElementById('crop-h');
const outputW = document.getElementById('crop-out-w');
const outputH = document.getElementById('crop-out-h');

// 初始化上传

  document.getElementById('crop-upload'),
  document.getElementById('crop-file'),
  async files => {

    if (!f || !f.type.startsWith('image/')) return;
    try {
      cropImg = await loadImage(f);
      initCropEditor();
      document.getElementById('crop-editor-card').style.display = '';
      showToast('图片已加载，可通过数值精确控制裁剪区域', 'success');
    } catch (e) { showToast('图片加载失败', 'error'); }
  }
);

function initCropEditor() {
  if (!cropImg) return;
  
  // 计算画布尺寸（适应容器）
  const maxW = cropContainer.clientWidth - 24 || 600;
  const scale = Math.min(1, maxW / cropImg.naturalWidth);
  cropCanvas.width = cropImg.naturalWidth * scale;
  cropCanvas.height = cropImg.naturalHeight * scale;
  cropCanvas._scale = scale;
  
  // 初始化裁剪区域为全图
  inputX.max = cropImg.naturalWidth;
  inputY.max = cropImg.naturalHeight;
  inputW.max = cropImg.naturalWidth;
  inputH.max = cropImg.naturalHeight;
  
  inputX.value = 0;
  inputY.value = 0;
  inputW.value = cropImg.naturalWidth;
  inputH.value = cropImg.naturalHeight;
  
  drawCropPreview();
}

function drawCropPreview() {
  if (!cropImg) return;
  
  const scale = cropCanvas._scale || 1;
  const x = parseInt(inputX.value) || 0;
  const y = parseInt(inputY.value) || 0;
  const w = parseInt(inputW.value) || 1;
  const h = parseInt(inputH.value) || 1;
  
  // 清空并绘制原图
  cropCtx.clearRect(0, 0, cropCanvas.width, cropCanvas.height);
  cropCtx.drawImage(cropImg, 0, 0, cropCanvas.width, cropCanvas.height);
  
  // 绘制半透明遮罩
  cropCtx.fillStyle = 'rgba(0, 0, 0, 0.5)';
  cropCtx.fillRect(0, 0, cropCanvas.width, cropCanvas.height);
  
  // 清除裁剪区域的遮罩，显示原图
  cropCtx.clearRect(x * scale, y * scale, w * scale, h * scale);
  cropCtx.drawImage(
    cropImg,
    x, y, w, h,  // 源区域
    x * scale, y * scale, w * scale, h * scale  // 目标区域
  );
  
  // 绘制裁剪框边框
  cropCtx.strokeStyle = '#f59e0b';
  cropCtx.lineWidth = 2;
  cropCtx.setLineDash([6, 4]);
  cropCtx.strokeRect(x * scale, y * scale, w * scale, h * scale);
  cropCtx.setLineDash([]);
  
  // 绘制网格线（三分线）
  cropCtx.strokeStyle = 'rgba(245, 158, 11, 0.4)';
  cropCtx.lineWidth = 1;
  const thirdW = (w * scale) / 3;
  const thirdH = (h * scale) / 3;
  for (let i = 1; i < 3; i++) {
    cropCtx.beginPath();
    cropCtx.moveTo(x * scale + thirdW * i, y * scale);
    cropCtx.lineTo(x * scale + thirdW * i, (y + h) * scale);
    cropCtx.stroke();
    cropCtx.beginPath();
    cropCtx.moveTo(x * scale, y * scale + thirdH * i);
    cropCtx.lineTo((x + w) * scale, y * scale + thirdH * i);
    cropCtx.stroke();
  }
  
  // 更新输出尺寸显示
  outputW.textContent = w;
  outputH.textContent = h;
}

// 输入框事件：约束值范围
function clampCropInput() {
  if (!cropImg) return;
  
  const imgW = cropImg.naturalWidth;
  const imgH = cropImg.naturalHeight;
  
  let x = Math.max(0, Math.min(parseInt(inputX.value) || 0, imgW - 1));
  let y = Math.max(0, Math.min(parseInt(inputY.value) || 0, imgH - 1));
  let w = Math.max(1, Math.min(parseInt(inputW.value) || 1, imgW - x));
  let h = Math.max(1, Math.min(parseInt(inputH.value) || 1, imgH - y));
  
  // 比例锁定：根据改变的是宽度还是高度来调整
  if (cropRatio > 0) {
    const targetW = h * cropRatio;
    const targetH = w / cropRatio;
    
    if (targetW <= imgW - x) {
      w = Math.round(targetW);
    } else {
      w = imgW - x;
      h = Math.round(w / cropRatio);
    }
  }
  
  inputX.value = x;
  inputY.value = y;
  inputW.value = w;
  inputH.value = h;
  
  drawCropPreview();
}

[inputX, inputY, inputW, inputH].forEach(input => {
  input.addEventListener('input', clampCropInput);
});

// 比例锁定切换
document.getElementById('crop-ratio').addEventListener('change', function() {
  cropRatio = parseFloat(this.value);
  if (cropRatio > 0) {
    lockIndicator.className = 'lock-indicator locked';
    lockIndicator.innerHTML = '<span>🔒</span> 已锁定';
    // 应用比例到当前宽度
    const w = parseInt(inputW.value);
    inputH.value = Math.round(w / cropRatio);
    clampCropInput();
    showToast(`比例已锁定为 ${this.options[this.selectedIndex].text}`, 'info');
  } else {
    lockIndicator.className = 'lock-indicator unlocked';
    lockIndicator.innerHTML = '<span>🔓</span> 未锁定';
  }
});

        break;
      case 'center-75':

        inputY.value = Math.round(imgH * 0.125);
        inputW.value = Math.round(imgW * 0.75);

        inputY.value = Math.round(imgH * 0.5);
        inputW.value = imgW;
        inputH.value = Math.round(imgH * 0.5);

    // 如果有比例锁定，重新应用
